struct

Agrupa variáveis de tipos diferentes sob um mesmo nome — base de todos os nodos

Declaração e uso
typedef struct Nodo {
    int         dado;
    struct Nodo *prox; /* auto-ref */
} Nodo;

Nodo  n;
n.dado    = 10;   /* operador . */

Nodo *p   = &n;
p->dado   = 20;   /* operador -> */
/* = (*p).dado  */
. vs ->
. acessa campo direto  |  -> acessa campo via ponteiro
p->dado é exatamente igual a (*p).dado
Nodo com auto-referência
typedef struct NodoDuplo {
    int dado;
    struct NodoDuplo *prox;
    struct NodoDuplo *ant;
} NodoDuplo;
A auto-referência usa struct Nodo — o typedef ainda não existe naquele ponto da declaração.
Pegadinhas de prova
Esquecer ; após } → erro de compilação
Usar . com ponteiro → erro de tipo
Usar -> com variável direta → erro de tipo

Ponteiro

Armazena o endereço de outra variável — & = endereço, * = valor no endereço

Operadores fundamentais
int  x = 42;
int *p = &x;      /* & = endereço */

printf("%d", *p); /* * = valor → 42 */
*p = 99;           /* altera x via p */

Nodo *n = (Nodo*) malloc(sizeof(Nodo));
if (!n) { /* trate erro! */ }
n->dado = 10;
free(n);  n = NULL;

Erros clássicos
Ponteiro não inicializado → segfault
free() duplo → comportamento indefinido
free() não zera o ponteiro — use p = NULL depois

Ponteiro de Ponteiro

Necessário para modificar ponteiros dentro de funções

Declaração e desreferência
int   x  = 10;
int  *p  = &x;   /* p → x */
int **pp = &p;   /* pp → p → x */

printf("%d", **pp); /* 10 */
**pp = 99;           /* altera x */
Por que ** em listas?
void insInicio(Nodo **ini, int v) {
    Nodo *novo = malloc(sizeof(Nodo));
    novo->dado = v;
    novo->prox = *ini;
    *ini = novo; /* modifica o ptr do chamador */
}
Nodo *lista = NULL;
insInicio(&lista, 10);

*pp = 0x100 (p)  |  **pp = 10 (x)

**pp desreferencia pp → p → x
Regra de leitura

Da direita para a esquerda:
int **pp = "pp é ponteiro para ponteiro para int"

Sem ** a função recebe CÓPIA do ponteiro — lista do chamador não muda.

Aritmética de Ponteiros

p+n avança n × sizeof(*p) bytes — não apenas 1 byte

Código e resultado
int arr[5] = {10,20,30,40,50};
int *p = arr;

printf("%d", *p);      /* 10 */
p++;
printf("%d", *p);      /* 20 */
printf("%d", *(p+2));  /* 40 */
printf("%d", p[1]);    /* 30 */

int *fim = arr + 5;
for(p=arr; p<fim; p++)
    printf("%d ", *p);
p[i] é açúcar sintático para *(p+i) — código idêntico.
Array na memória — arraste o ponteiro

p aponta para arr[0] = 10

p aponta para arr[0]. *p = 10
Regras fundamentais
  • p+n avança n × sizeof(*p) bytes
  • p-q retorna nº de elementos entre eles
  • arr++ é inválido — arr é constante
  • Não acesse além de arr+n (UB)

Pilha — LIFO

LIFO Last In, First Out · Inserção e remoção sempre pelo topo

Empilhar
void empilhar(Pilha *p, int v) {
    Nodo *n = malloc(sizeof(Nodo));
    n->dado = v;
    n->prox = p->topo; /* ANTES */
    p->topo = n;        /* DEPOIS */
    p->tam++;
}
Inverter a ordem cria ciclo — nodo aponta para si mesmo.
Desempilhar
int desempilhar(Pilha *p) {
    if (!p->topo) return -1;
    Nodo *tmp = p->topo;
    int   val = tmp->dado;
    p->topo   = p->topo->prox;
    free(tmp); p->tam--;
    return val;
}
Visualizador interativo
Pilha vazia — empilhe um elemento!
Complexidade
empilhar()O(1)
desempilhar()O(1)
exibir()O(n)
Aplicações: recursão, undo/redo, DFS, verificação de parênteses.

Fila — FIFO

FIFO First In, First Out · Frente remove, fim insere

Enfileirar + Desenfileirar
void enfileirar(Fila *f, int v) {
    Nodo *n=malloc(sizeof(Nodo));
    n->dado=v; n->prox=NULL;
    if(!f->frente) f->frente=f->fim=n;
    else{ f->fim->prox=n; f->fim=n; }
    f->tam++;
}
int desenfileirar(Fila *f) {
    if(!f->frente) return -1;
    Nodo *tmp=f->frente;
    int v=tmp->dado;
    f->frente=f->frente->prox;
    if(!f->frente) f->fim=NULL; /* ! */
    free(tmp); f->tam--;
    return v;
}
Visualizador interativo
Fila vazia
frente→NULL · fim→NULL · tam=0
Aguardando...
Caso especial: ao remover o último elemento, fim TAMBÉM deve virar NULL. Esquecer → segfault na próxima inserção.
enfileirar()O(1)
desenfileirar()O(1)
exibir()O(n)

Lista Simplesmente Encadeada

Generaliza Pilha e Fila · Inserção e remoção em qualquer posição

Inserir no início — O(1)
void insInicio(Nodo **ini, int v){
    Nodo *n=malloc(sizeof(Nodo));
    n->dado=v; n->prox=*ini;
    *ini=n;
}
Remover por valor — O(n)
void remover(Nodo **ini, int v){
    Nodo *a=NULL, *c=*ini;
    while(c && c->dado!=v){a=c;c=c->prox;}
    if(!c) return;
    if(!a) *ini=c->prox;
    else   a->prox=c->prox;
    free(c);
}
Dois ponteiros: anterior (a) e atual (c). Sem anterior não é possível religar a cadeia.
Inverter — O(n) tempo, O(1) espaço
void inverter(Nodo **ini){
    Nodo *a=NULL,*c=*ini,*prx;
    while(c){
        prx=c->prox;
        c->prox=a; a=c; c=prx;
    }
    *ini=a;
}

Lista Duplamente Encadeada

prox + ant · Remoção por ponteiro em O(1) · Percurso reverso

Estrutura
typedef struct ND {
    int dado;
    struct ND *prox;
    struct ND *ant;
} ND;
Remover por ponteiro — O(1)
void remover(ND **ini, ND *n){
    if(n->ant) n->ant->prox=n->prox;
    else       *ini=n->prox;
    if(n->prox) n->prox->ant=n->ant;
    free(n);
}
Dado o ponteiro, remove em O(1). Na lista simples seria O(n).
Percurso reverso
void exibirRev(ND *fim){
    while(fim){
        printf("%d ", fim->dado);
        fim=fim->ant;
    }
}
